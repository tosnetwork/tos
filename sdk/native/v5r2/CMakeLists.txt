cmake_minimum_required(VERSION 3.16)
project(tos_v5r2_native LANGUAGES C CXX)
get_filename_component(TOS_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ML "${TOS_SOURCE_ROOT}/third-party/mldsa-native/mldsa")
set(SLH "${TOS_SOURCE_ROOT}/third-party/slhdsa-c")
add_library(tos_v5r2_backends STATIC "${ML}/mldsa_native.c" "${SLH}/slh_dsa.c"
  "${SLH}/slh_sha2.c" "${SLH}/sha2_256.c" "${SLH}/sha2_512.c")
set_target_properties(tos_v5r2_backends PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_v5r2_backends PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ML}" "${SLH}")
target_compile_definitions(tos_v5r2_backends PRIVATE MLD_CONFIG_FILE="mldsa-config.h")
add_library(tos_v5r2_native STATIC tos_v5r2.c tos_v5r2_kdf.c)
set_target_properties(tos_v5r2_native PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_v5r2_native PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${ML}" "${SLH}")
target_compile_definitions(tos_v5r2_native PRIVATE MLD_CONFIG_FILE="mldsa-config.h" TOS_V5R2_ML_PREFIX=tos_v5r2_mldsa44)
target_link_libraries(tos_v5r2_native PUBLIC tos_v5r2_backends)
add_library(tos_v5r2_fee_verify STATIC
  "${TOS_SOURCE_ROOT}/crypto/pq/lms-fee.cpp"
  "${TOS_SOURCE_ROOT}/crypto/pq/wallet-lms-fee-c.cpp")
set_target_properties(tos_v5r2_fee_verify PROPERTIES CXX_STANDARD 17
  CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
target_compile_definitions(tos_v5r2_fee_verify PRIVATE TOS_LMS_PORTABLE_SHA256=1)
target_include_directories(tos_v5r2_fee_verify PRIVATE "${SLH}")
target_link_libraries(tos_v5r2_fee_verify PUBLIC tos_v5r2_backends)
set(LMS "${TOS_SOURCE_ROOT}/third-party/lms-reference")
add_library(tos_v5r2_lms_primitives STATIC "${LMS}/hss_derive.c" "${LMS}/hss_zeroize.c"
  "${LMS}/lm_common.c" "${LMS}/lm_ots_common.c" "${LMS}/lm_ots_sign.c"
  "${LMS}/endian.c" "${LMS}/hash.c")
set_target_properties(tos_v5r2_lms_primitives PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_v5r2_lms_primitives PRIVATE "${LMS}" "${SLH}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tos_v5r2_lms_primitives PRIVATE EXT_SHA256_H="tos_lms_sha256.h")
target_link_libraries(tos_v5r2_lms_primitives PUBLIC tos_v5r2_backends)
add_library(tos_v5r2_fee_sign STATIC "${TOS_SOURCE_ROOT}/crypto/pq/wallet-lms-sign-c.cpp")
set_target_properties(tos_v5r2_fee_sign PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_v5r2_fee_sign PRIVATE "${LMS}" "${SLH}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tos_v5r2_fee_sign PRIVATE TOS_LMS_PORTABLE_SHA256=1)
target_link_libraries(tos_v5r2_fee_sign PUBLIC tos_v5r2_lms_primitives tos_v5r2_fee_verify)
enable_testing()
add_executable(test-v5r2-mobile-lms-sign "${TOS_SOURCE_ROOT}/test/wallet-v5r2/mobile-lms-sign-test.c")
target_link_libraries(test-v5r2-mobile-lms-sign PRIVATE tos_v5r2_fee_sign)
add_test(NAME v5r2-mobile-lms-sign COMMAND test-v5r2-mobile-lms-sign)
add_executable(test-v5r2-mobile-kdf "${TOS_SOURCE_ROOT}/test/wallet-v5r2/mobile-kdf-test.c")
target_link_libraries(test-v5r2-mobile-kdf PRIVATE tos_v5r2_native)
add_test(NAME v5r2-mobile-kdf COMMAND test-v5r2-mobile-kdf)
add_executable(test-v5r2-mobile-lms "${TOS_SOURCE_ROOT}/test/wallet-v5r2/mobile-lms-verify-test.c")
target_link_libraries(test-v5r2-mobile-lms PRIVATE tos_v5r2_fee_verify)
add_test(NAME v5r2-mobile-lms COMMAND test-v5r2-mobile-lms)
add_executable(test-v5r2-mobile-native "${TOS_SOURCE_ROOT}/test/wallet-v5r2/mobile-native-test.c")
target_link_libraries(test-v5r2-mobile-native PRIVATE tos_v5r2_native)
target_include_directories(test-v5r2-mobile-native PRIVATE "${ML}" "${SLH}")
target_compile_definitions(test-v5r2-mobile-native PRIVATE MLD_CONFIG_FILE="mldsa-config.h")
add_test(NAME v5r2-mobile-native COMMAND test-v5r2-mobile-native)
