cmake_minimum_required(VERSION 3.16)
project(tos_quantum_native LANGUAGES C CXX)
get_filename_component(TOS_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ML "${TOS_SOURCE_ROOT}/third-party/mldsa-native/mldsa")
set(SLH "${TOS_SOURCE_ROOT}/third-party/slhdsa-c")
add_library(tos_quantum_backends STATIC "${ML}/mldsa_native.c" "${SLH}/slh_dsa.c"
  "${SLH}/slh_sha2.c" "${SLH}/sha2_256.c" "${SLH}/sha2_512.c")
set_target_properties(tos_quantum_backends PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_quantum_backends PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ML}" "${SLH}")
target_compile_definitions(tos_quantum_backends PRIVATE MLD_CONFIG_FILE="mldsa-config.h")
add_library(tos_quantum_native STATIC tos_quantum.c tos_quantum_kdf.c)
set_target_properties(tos_quantum_native PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_quantum_native PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${ML}" "${SLH}")
target_compile_definitions(tos_quantum_native PRIVATE MLD_CONFIG_FILE="mldsa-config.h" TOS_Quantum_ML_PREFIX=tos_quantum_mldsa44)
target_link_libraries(tos_quantum_native PUBLIC tos_quantum_backends)
add_library(tos_quantum_fee_verify STATIC
  "${TOS_SOURCE_ROOT}/crypto/pq/lms-fee.cpp"
  "${TOS_SOURCE_ROOT}/crypto/pq/wallet-lms-fee-c.cpp")
set_target_properties(tos_quantum_fee_verify PROPERTIES CXX_STANDARD 17
  CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
target_compile_definitions(tos_quantum_fee_verify PRIVATE TOS_LMS_PORTABLE_SHA256=1)
target_include_directories(tos_quantum_fee_verify PRIVATE "${SLH}")
target_link_libraries(tos_quantum_fee_verify PUBLIC tos_quantum_backends)
set(LMS "${TOS_SOURCE_ROOT}/third-party/lms-reference")
add_library(tos_quantum_lms_primitives STATIC "${LMS}/hss_derive.c" "${LMS}/hss_zeroize.c"
  "${LMS}/lm_common.c" "${LMS}/lm_ots_common.c" "${LMS}/lm_ots_sign.c"
  "${LMS}/endian.c" "${LMS}/hash.c")
set_target_properties(tos_quantum_lms_primitives PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_quantum_lms_primitives PRIVATE "${LMS}" "${SLH}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tos_quantum_lms_primitives PRIVATE EXT_SHA256_H="tos_lms_sha256.h")
target_link_libraries(tos_quantum_lms_primitives PUBLIC tos_quantum_backends)
add_library(tos_quantum_fee_sign STATIC "${TOS_SOURCE_ROOT}/crypto/pq/wallet-lms-sign-c.cpp")
set_target_properties(tos_quantum_fee_sign PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
target_include_directories(tos_quantum_fee_sign PRIVATE "${LMS}" "${SLH}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tos_quantum_fee_sign PRIVATE TOS_LMS_PORTABLE_SHA256=1)
target_link_libraries(tos_quantum_fee_sign PUBLIC tos_quantum_lms_primitives tos_quantum_fee_verify)
enable_testing()
add_executable(test-quantum-mobile-lms-sign "${TOS_SOURCE_ROOT}/test/wallet-quantum/mobile-lms-sign-test.c")
target_link_libraries(test-quantum-mobile-lms-sign PRIVATE tos_quantum_fee_sign)
add_test(NAME quantum-mobile-lms-sign COMMAND test-quantum-mobile-lms-sign)
add_executable(test-quantum-mobile-kdf "${TOS_SOURCE_ROOT}/test/wallet-quantum/mobile-kdf-test.c")
target_link_libraries(test-quantum-mobile-kdf PRIVATE tos_quantum_native)
add_test(NAME quantum-mobile-kdf COMMAND test-quantum-mobile-kdf)
add_executable(test-quantum-mobile-lms "${TOS_SOURCE_ROOT}/test/wallet-quantum/mobile-lms-verify-test.c")
target_link_libraries(test-quantum-mobile-lms PRIVATE tos_quantum_fee_verify)
add_test(NAME quantum-mobile-lms COMMAND test-quantum-mobile-lms)
add_executable(test-quantum-mobile-native "${TOS_SOURCE_ROOT}/test/wallet-quantum/mobile-native-test.c")
target_link_libraries(test-quantum-mobile-native PRIVATE tos_quantum_native)
target_include_directories(test-quantum-mobile-native PRIVATE "${ML}" "${SLH}")
target_compile_definitions(test-quantum-mobile-native PRIVATE MLD_CONFIG_FILE="mldsa-config.h")
add_test(NAME quantum-mobile-native COMMAND test-quantum-mobile-native)
